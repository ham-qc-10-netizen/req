    function updateProgressTimers() {
      const now = new Date();
      currentOrders.forEach(o => {
        const created = new Date(o.createdAt);
        const printed = o.printedAt ? new Date(o.printedAt) : null;
        const delivered = o.deliveredAt ? new Date(o.deliveredAt) : null;

        const f1Bar = document.getElementById(`bar-f1-${o.id}`);
        const f2Bar = document.getElementById(`bar-f2-${o.id}`);
        const f3Bar = document.getElementById(`bar-f3-${o.id}`);
        const label = document.getElementById(`label-fase-${o.id}`);
        const timer = document.getElementById(`timer-${o.id}`);

        if (!f1Bar || !f2Bar || !f3Bar || !label || !timer) return;

        if (delivered) {
          f1Bar.style.width = '100%';
          f2Bar.style.width = '100%';
          f3Bar.style.width = '100%';
          label.textContent = `Entregado en Línea (Reloj: ${o.deliveredBy})`;
          timer.textContent = formatSec(o.metrics.lead_time_total_sec || 0);
        } else if (printed) {
          f1Bar.style.width = '100%';
          const elapsedSec = Math.floor((now - printed) / 1000);
          if (elapsedSec <= 300) {
            f2Bar.style.width = `${Math.min(100, (elapsedSec / 300) * 100)}%`;
            f3Bar.style.width = '0%';
            label.textContent = 'F2: Recolección en Almacén';
            timer.textContent = formatSec(300 - elapsedSec);
          } else {
            f2Bar.style.width = '100%';
            const transitSec = elapsedSec - 300;
            f3Bar.style.width = `${Math.min(100, (transitSec / 300) * 100)}%`;
            label.textContent = 'F3: En Tránsito a Línea';
            timer.textContent = formatSec(Math.max(0, 300 - transitSec));
          }
        } else {
          const elapsedSec = Math.floor((now - created) / 1000);
          f1Bar.style.width = `${Math.min(100, (elapsedSec / 180) * 100)}%`;
          f2Bar.style.width = '0%';
          f3Bar.style.width = '0%';
          label.textContent = 'F1: Tolerancia / Revisión';
          timer.textContent = formatSec(Math.max(0, 180 - elapsedSec));
        }
      });
    }

    function formatSec(seconds) {
      const m = Math.floor(seconds / 60);
      const s = seconds % 60;
      return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    }

      if (!order.printedAt) {

        order.status = 'En Proceso';

      }

      cargarReporteEnPantalla(order);
      switchView('reporte');
      renderOrdersTable();
      updateKPIs();

      setTimeout(() => { window.print(); }, 400);
    }

      document.getElementById('rep-total-qty').textContent = order.totalPieces;

      if (order.deliveredAt) {
        document.getElementById('rep-sign-linea').innerHTML = `<span class="text-emerald-700 font-bold text-lg font-signature">Validado Reloj #${order.deliveredBy}</span>`;
        document.getElementById('rep-reloj-tag').textContent = `Recibido: Reloj #${order.deliveredBy}`;
        document.getElementById('rep-lead-time').textContent = `${Math.round((order.metrics.lead_time_total_sec || 0)/60)} min`;
      } else {
        document.getElementById('rep-sign-linea').innerHTML = `<span class="text-slate-300 italic text-xs font-sans">Pendiente Entrega</span>`;
        document.getElementById('rep-reloj-tag').textContent = `Recibido en Línea`;
        document.getElementById('rep-lead-time').textContent = `--:-- min`;
      }
    }

    function abrirModalEntrega() {
      document.getElementById('modal-entrega').classList.remove('hidden');
      document.getElementById('input-reloj').value = '';
      document.getElementById('error-reloj').classList.add('hidden');
      document.getElementById('input-reloj').focus();
    }

    function cerrarModalEntrega() {
      document.getElementById('modal-entrega').classList.add('hidden');
    }

    async function ejecutarEntregaReloj() {
      const badgeInput = document.getElementById('input-reloj').value.trim();
      const errorP = document.getElementById('error-reloj');

      if (!AUTHORIZED_BADGES.includes(badgeInput)) {
        errorP.classList.remove('hidden');
        return;
      }

      const order = currentOrders.find(o => o.id === activeOrderIdForDelivery) || currentOrders[currentOrders.length - 1];
      if (!order) {
        alert('No se encontró orden activa para entregar.');
        cerrarModalEntrega();
        return;
      }

      try {
        await fetch(`/api/orders/${order.id}/deliver`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ badge: badgeInput })
        });
      } catch (e) {}

      const now = new Date();
      order.deliveredAt = now.toISOString();
      order.deliveredBy = badgeInput;
      order.status = 'Entregado';
      order.metrics.lead_time_total_sec = Math.floor((now - new Date(order.createdAt)) / 1000);

      localStorage.setItem('orders_db', JSON.stringify(currentOrders));

      cerrarModalEntrega();
      document.getElementById('mx5a-delivery-box').classList.add('hidden');

      renderOrdersTable();
      renderHistorialTable();
      updateKPIs();
      cargarReporteEnPantalla(order);

      alert(`¡Entrega confirmada con éxito por Reloj #${badgeInput}! Ciclo cerrado.`);
    }

    function renderHistorialTable() {
      const tbody = document.getElementById('historial-tbody');
      tbody.innerHTML = '';

      if (currentOrders.length === 0) {
        tbody.innerHTML = '<tr><td colspan="8" class="py-6 text-center text-slate-400 font-sans">No hay historial registrado aún.</td></tr>';
        return;
      }

      currentOrders.slice().reverse().forEach(o => {
        const tr = document.createElement('tr');
        tr.className = "hover:bg-slate-50 border-b border-slate-100";
        tr.innerHTML = `
          <td class="py-2.5 px-4 font-bold text-blue-700">${o.id}</td>
          <td class="py-2.5 px-4 font-sans font-semibold">${o.model}</td>
          <td class="py-2.5 px-4">${o.totalPieces} pzs</td>
          <td class="py-2.5 px-4 text-slate-500">${new Date(o.createdAt).toLocaleTimeString()}</td>
          <td class="py-2.5 px-4 text-slate-500">${o.printedAt ? new Date(o.printedAt).toLocaleTimeString() : '--'}</td>
          <td class="py-2.5 px-4 text-slate-500">${o.deliveredAt ? new Date(o.deliveredAt).toLocaleTimeString() : '--'}</td>
          <td class="py-2.5 px-4 font-bold text-emerald-700">${o.deliveredBy ? '#' + o.deliveredBy : 'Pendiente'}</td>
          <td class="py-2.5 px-4 font-bold text-slate-800">${o.metrics.lead_time_total_sec ? Math.round(o.metrics.lead_time_total_sec/60) + ' min' : '--'}</td>
        `;
        tbody.appendChild(tr);
      });
    }

    function updateKPIs() {
      const pendientes = currentOrders.filter(o => o.status === 'Pendiente').length;
      const proceso = currentOrders.filter(o => o.status === 'En Proceso').length;
      const completados = currentOrders.filter(o => o.status === 'Entregado').length;

      document.getElementById('kpi-pendientes').textContent = pendientes;
      document.getElementById('kpi-proceso').textContent = proceso;
      document.getElementById('kpi-completados').textContent = completados;

      const badgeNav = document.getElementById('badge-pendientes-nav');
      if (pendientes > 0) {
        badgeNav.textContent = pendientes;
        badgeNav.classList.remove('hidden');
      } else {
        badgeNav.classList.add('hidden');
      }
    }

    function exportarCSVLocal() {
      if (currentOrders.length === 0) {
        alert('No hay datos en el historial para exportar.');
        return;
      }
      let csv = "\uFEFF";
      csv += "Folio,Modelo,Estatus,Partes,Total Piezas,Solicitado,Impreso,Entregado,No Reloj,Lead Time (seg)\n";
      currentOrders.forEach(o => {
        csv += `"${o.id}","${o.model}","${o.status}","${o.items.length}","${o.totalPieces}","${o.createdAt}","${o.printedAt || ''}","${o.deliveredAt || ''}","${o.deliveredBy || ''}","${o.metrics.lead_time_total_sec || ''}"\n`;
      });
      const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
      const link = document.createElement("a");
      link.href = URL.createObjectURL(blob);
      link.download = `requisiciones_mx5a_${new Date().toISOString().slice(0,10)}.csv`;
      link.click();
    }
